const HOJA_PARTIDAS    = 'partidas';
const HOJA_COMPARATIVA = 'comparativa';
const HOJA_RESUMEN     = 'resumen';

/* Una fila por CADA VEZ que alguien juega: se crea al registrarse y se
   va actualizando en cada etapa (misma sesión = misma fila, por su ID). */
const CAB_PARTIDAS = ['Fecha','Nombre','Apellido','Sección','Etapa','Partidas','Puntaje','Llaves','Tiempo','Terminó','Código','ID'];
/* Una fila por ESTUDIANTE (aunque escriba su nombre distinto cada vez). */
const CAB_RESUMEN  = ['Nombre','Apellido','Sección','Sesiones','Escapó','Último código','Mejor puntaje','Máx. llaves','Tiempo total','Última vez'];

/* ---------------- entrada ---------------- */
function doPost(e) {
  const lock = LockService.getScriptLock();
  lock.waitLock(30000);
  try {
    const d = JSON.parse(e.postData.contents);
    const libro = SpreadsheetApp.getActiveSpreadsheet();
    const partidas = hoja_(libro, HOJA_PARTIDAS, CAB_PARTIDAS);
    hojaVacia_(libro, HOJA_COMPARATIVA);      // se crea antes que "resumen" para quedar como hoja 2

    upsertPartida_(partidas, d);

    const filas  = leerPartidas_(partidas);
    const grupos = agrupar_(filas);
    escribirResumen_(libro, grupos);
    escribirComparativa_(libro.getSheetByName(HOJA_COMPARATIVA), filas, grupos);
    return salida_('ok');
  } finally {
    lock.releaseLock();
  }
}

// para probar que la URL funciona: abrirla en el navegador muestra este mensaje
function doGet() {
  return salida_('Escape Celular: registro activo');
}

/* ---------------- partidas ----------------
   La misma sesión de juego (mismo ID) actualiza SIEMPRE la misma fila:
   así la Etapa avanza (Registrado → Niveles → … → ¡ESCAPÓ!) sin duplicar. */
function upsertPartida_(hoja, d) {
  const rid = String(d.rid || '');
  const fila = [
    new Date(),
    String(d.nombre || ''), String(d.apellido || ''), String(d.seccion || ''),
    String(d.etapa || ''), Number(d.partidas) || 0, Number(d.puntaje) || 0,
    Number(d.llaves) || 0, fmtTiempo_(Number(d.tiempo_seg) || 0),
    String(d.termino || 'No'), String(d.codigo || ''), rid
  ];
  if (rid) {
    const filaExistente = buscarRid_(hoja, rid);
    if (filaExistente > 0) {
      // no pisa un ¡ESCAPÓ! con una etapa anterior (reintentos fuera de orden)
      const etapaAnterior = String(hoja.getRange(filaExistente, 5).getValue());
      if (etapaAnterior === '¡ESCAPÓ!' && d.etapa !== '¡ESCAPÓ!') return;
      hoja.getRange(filaExistente, 1, 1, CAB_PARTIDAS.length).setValues([fila]);
      return;
    }
  }
  hoja.appendRow(fila);
}

function buscarRid_(hoja, rid) {
  const ultima = hoja.getLastRow();
  if (ultima < 2) return 0;
  const desde = Math.max(2, ultima - 499);    // revisa las últimas 500 filas
  const ids = hoja.getRange(desde, CAB_PARTIDAS.length, ultima - desde + 1, 1).getValues();
  for (let i = ids.length - 1; i >= 0; i--) if (String(ids[i][0]) === rid) return desde + i;
  return 0;
}

function leerPartidas_(partidas) {
  const ultima = partidas.getLastRow();
  if (ultima < 2) return [];
  return partidas.getRange(2, 1, ultima - 1, CAB_PARTIDAS.length).getValues();
}

/* ---------------- hojas ---------------- */
function hoja_(libro, nombre, cabeceras) {
  let h = libro.getSheetByName(nombre);
  if (!h) {
    // si es la planilla recién creada (una sola hoja vacía "Hoja 1"), la renombra:
    // así "partidas" queda como hoja 1, como pidió la maestra
    const todas = libro.getSheets();
    if (nombre === HOJA_PARTIDAS && todas.length === 1 && todas[0].getLastRow() === 0) {
      h = todas[0];
      h.setName(nombre);
    } else {
      h = libro.insertSheet(nombre);
    }
  }
  if (h.getLastRow() === 0) {
    h.getRange(1, 1, 1, cabeceras.length).setValues([cabeceras]).setFontWeight('bold');
    h.setFrozenRows(1);
  }
  return h;
}

function hojaVacia_(libro, nombre) {
  let h = libro.getSheetByName(nombre);
  if (!h) h = libro.insertSheet(nombre);
  return h;
}

function salida_(txt) {
  return ContentService.createTextOutput(txt).setMimeType(ContentService.MimeType.TEXT);
}

/* ---------------- tiempos ---------------- */
function fmtTiempo_(seg) {
  seg = Math.round(seg);
  const m = Math.floor(seg / 60), s = seg % 60;
  return m + 'm ' + (s < 10 ? '0' : '') + s + 's';
}
function minTot_(seg) {
  const m = Math.round(seg / 60);
  return m >= 60 ? Math.floor(m / 60) + 'h ' + (m % 60) + 'm' : m + 'm';
}
function segDeTexto_(txt) {
  const m = /(\d+)m\s*(\d+)s/.exec(String(txt || ''));
  return m ? (+m[1]) * 60 + (+m[2]) : 0;
}

/* ---------------- normalización de nombres ----------------
   Ignora mayúsculas, tildes y espacios dobles:
   "EMMA  García", "emma garcia" y "Emma García" son la misma jugadora. */
function norm_(s) {
  return String(s || '').toLowerCase()
    .normalize('NFD').replace(/[̀-ͯ]/g, '')
    .replace(/\s+/g, ' ').trim();
}
function normSeccion_(s) { return norm_(s).replace(/[^a-z0-9]/g, ''); }   // "1° A" -> "1a"

/* distancia de edición: detecta que "ema" y "emma" son la misma persona
   (una letra de más, de menos o cambiada) */
function dist_(a, b) {
  const m = a.length, n = b.length;
  if (Math.abs(m - n) > 2) return 99;
  const d = [];
  for (let i = 0; i <= m; i++) d[i] = [i];
  for (let j = 0; j <= n; j++) d[0][j] = j;
  for (let i = 1; i <= m; i++)
    for (let j = 1; j <= n; j++)
      d[i][j] = Math.min(d[i-1][j] + 1, d[i][j-1] + 1, d[i-1][j-1] + (a[i-1] === b[j-1] ? 0 : 1));
  return d[m][n];
}

function mismoEstudiante_(x, y) {
  if (x.sec !== y.sec) return false;                       // secciones distintas: nunca es la misma
  if (x.clave === y.clave) return true;                    // nombre+apellido idénticos (normalizados)
  return dist_(x.clave, y.clave) <= 1;                     // una letra de diferencia: misma persona
}

/* ---------------- agrupar filas por estudiante ---------------- */
function agrupar_(filas) {
  const grupos = [];
  for (const f of filas) {
    const n = norm_(f[1]), a = norm_(f[2]), sec = normSeccion_(f[3]);
    if (!n && !a) continue;
    const yo = { sec, clave: n + ' ' + a };
    let g = null;
    for (const cand of grupos) {
      if (mismoEstudiante_({ sec: cand.sec, clave: cand.clave }, yo)) { g = cand; break; }
    }
    if (!g) {
      g = { sec, clave: yo.clave,
            nomRaw: String(f[1]).trim(), apeRaw: String(f[2]).trim(), secRaw: String(f[3]).trim(),
            sesiones: 0, escapes: 0, ultCodigo: '', maxPuntaje: 0, maxLlaves: 0,
            tiempoSeg: 0, ultima: null };
      grupos.push(g);
    } else {
      // si escribió el nombre más completo otra vez ("emma" sobre "ema"), se conserva la forma larga
      if (norm_(f[1] + ' ' + f[2]).length > norm_(g.nomRaw + ' ' + g.apeRaw).length) {
        g.nomRaw = String(f[1]).trim(); g.apeRaw = String(f[2]).trim();
      }
    }
    g.sesiones++;
    if (String(f[9]) === 'Sí') g.escapes++;
    if (String(f[10])) g.ultCodigo = String(f[10]);
    g.maxPuntaje = Math.max(g.maxPuntaje, Number(f[6]) || 0);
    g.maxLlaves  = Math.max(g.maxLlaves,  Number(f[7]) || 0);
    g.tiempoSeg += segDeTexto_(f[8]);
    const fecha = f[0] instanceof Date ? f[0] : new Date(f[0]);
    if (!g.ultima || fecha > g.ultima) { g.ultima = fecha; if (String(f[10])) g.ultCodigo = String(f[10]); }
  }
  return grupos;
}

/* ---------------- resumen por estudiante ---------------- */
function escribirResumen_(libro, grupos) {
  const resumen = hoja_(libro, HOJA_RESUMEN, CAB_RESUMEN);
  if (resumen.getLastRow() > 1)
    resumen.getRange(2, 1, resumen.getLastRow() - 1, CAB_RESUMEN.length).clearContent();

  grupos.sort((a, b) =>
    a.secRaw.localeCompare(b.secRaw) ||
    a.apeRaw.localeCompare(b.apeRaw) ||
    a.nomRaw.localeCompare(b.nomRaw));

  const filasR = grupos.map(g => [
    g.nomRaw, g.apeRaw, g.secRaw,
    g.sesiones, g.escapes, g.ultCodigo,
    g.maxPuntaje, g.maxLlaves, minTot_(g.tiempoSeg), g.ultima
  ]);
  if (filasR.length) resumen.getRange(2, 1, filasR.length, CAB_RESUMEN.length).setValues(filasR);
}

/* ---------------- comparativa entre secciones ----------------
   Una tabla "Métrica | Sección A | Sección B | ..." que se reescribe
   entera con cada partida: sirve para comparar cursos de un vistazo. */
function armarComparativa_(filas, grupos) {
  // estadísticas por sección a partir de las sesiones
  const secs = {};   // secNorm -> stats
  const asegurar = sec => {
    if (!secs[sec]) secs[sec] = { display: sec, partidas: 0, tiempoSeg: 0, punt: 0, maxPunt: 0,
      llaves: 0, escapes: 0, tiempoEscSeg: 0, mejorEsc: null, sinSalir: {}, est: 0, term: 0 };
    return secs[sec];
  };
  for (const f of filas) {
    const sec = normSeccion_(f[3]);
    if (!sec) continue;
    const s = asegurar(sec);
    if (String(f[3]).trim()) s.display = String(f[3]).trim();
    s.partidas++;
    const t = segDeTexto_(f[8]);
    s.tiempoSeg += t;
    const punt = Number(f[6]) || 0;
    s.punt += punt;
    s.maxPunt = Math.max(s.maxPunt, punt);
    s.llaves += Number(f[7]) || 0;
    if (String(f[9]) === 'Sí') {
      s.escapes++;
      s.tiempoEscSeg += t;
      if (s.mejorEsc === null || t < s.mejorEsc) s.mejorEsc = t;
    } else {
      const et = String(f[4] || 'Registrado').replace(/^Abandonó en /, '');
      s.sinSalir[et] = (s.sinSalir[et] || 0) + 1;
    }
  }
  // estudiantes únicos y cuántos terminaron (desde los grupos ya difusos)
  for (const g of grupos) {
    const s = asegurar(g.sec);
    s.est++;
    if (g.escapes > 0) s.term++;
  }

  const orden = Object.keys(secs).sort();
  const pct = (n, tot) => tot ? n + ' (' + Math.round(n * 100 / tot) + '%)' : String(n);
  const prom1 = (n, tot) => tot ? String(Math.round(n * 10 / tot) / 10) : '—';
  const moda = obj => {
    let mejor = '', mx = 0;
    for (const k in obj) if (obj[k] > mx) { mx = obj[k]; mejor = k; }
    return mejor ? mejor + ' (' + mx + ')' : '—';
  };
  const col = fn => orden.map(sec => fn(secs[sec]));

  const tabla = [
    ['Métrica'].concat(orden.map(sec => 'Sección ' + secs[sec].display)),
    ['Estudiantes que jugaron'].concat(col(s => s.est)),
    ['Partidas jugadas'].concat(col(s => s.partidas)),
    ['Partidas por estudiante'].concat(col(s => prom1(s.partidas, s.est))),
    ['Terminaron el juego'].concat(col(s => pct(s.term, s.est))),
    ['Participaron sin terminar'].concat(col(s => pct(s.est - s.term, s.est))),
    ['Escapes totales'].concat(col(s => s.escapes)),
    ['Tiempo promedio por partida'].concat(col(s => s.partidas ? fmtTiempo_(s.tiempoSeg / s.partidas) : '—')),
    ['Tiempo promedio hasta escapar'].concat(col(s => s.escapes ? fmtTiempo_(s.tiempoEscSeg / s.escapes) : '—')),
    ['Mejor tiempo hasta escapar'].concat(col(s => s.mejorEsc !== null ? fmtTiempo_(s.mejorEsc) : '—')),
    ['Puntaje promedio por partida'].concat(col(s => prom1(s.punt, s.partidas))),
    ['Mejor puntaje'].concat(col(s => s.maxPunt)),
    ['Llaves promedio por partida'].concat(col(s => prom1(s.llaves, s.partidas))),
    ['Etapa donde más se quedan sin terminar'].concat(col(s => moda(s.sinSalir))),
    [],
    ['Actualizado', new Date()]
  ];
  return { tabla, ncols: orden.length + 1 };
}

function escribirComparativa_(comp, filas, grupos) {
  const { tabla, ncols } = armarComparativa_(filas, grupos);
  comp.clearContents();
  comp.getRange(1, 1, tabla.length, ncols).setValues(tabla);
  comp.getRange(1, 1, 1, ncols).setFontWeight('bold');
  comp.getRange(1, 1, tabla.length - 2, 1).setFontWeight('bold');
  comp.setFrozenRows(1);
  comp.autoResizeColumns(1, ncols);
}
